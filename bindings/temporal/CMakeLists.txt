cmake_minimum_required(VERSION 3.24)
project(RexTemporalBindings VERSION 0.2.0 LANGUAGES CXX)
if(CMAKE_CXX_COMPILER_ID MATCHES "^(GNU|Clang|AppleClang)$")
  add_compile_options("-ffile-prefix-map=${CMAKE_CURRENT_BINARY_DIR}=."
                      "-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=.")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter Development.Module)
find_package(RexTemporal 0.1.0 EXACT CONFIG REQUIRED)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/cmake/CompanionReceipt.cmake")
  include(cmake/CompanionReceipt.cmake)
else()
  include(../cpp/cmake/CompanionReceipt.cmake)
endif()
rexgraph_companion_receipt(sdk_digest sdk_members temporal_sdk_include)
rexgraph_install_companion_terms(rexgraph_temporal)
set(temporal_receipt "sdk:${sdk_digest}\n")
foreach(source temporal_bridge.hpp rexgraph_temporal/_native.pyx)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${source}" digest)
  string(APPEND temporal_receipt "binding/${source}:${digest}\n")
endforeach()
string(SHA256 temporal_digest "${temporal_receipt}")
set(generated "${CMAKE_CURRENT_BINARY_DIR}/_native.cpp")
add_custom_command(OUTPUT "${generated}"
  COMMAND "${Python3_EXECUTABLE}" -m cython --cplus -3
          --output-file "${generated}" "${CMAKE_CURRENT_SOURCE_DIR}/rexgraph_temporal/_native.pyx"
  DEPENDS rexgraph_temporal/_native.pyx temporal_bridge.hpp VERBATIM)
Python3_add_library(_native MODULE WITH_SOABI "${generated}")
target_compile_features(_native PRIVATE cxx_std_20)
target_include_directories(_native PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(_native PRIVATE rex_temporal::runtime)
target_compile_definitions(_native PRIVATE REX_TEMPORAL_DIGEST="${temporal_digest}")
install(TARGETS _native LIBRARY DESTINATION rexgraph_temporal)
